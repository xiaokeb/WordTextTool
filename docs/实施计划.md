# Word 文本处理工具首版实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 构建一个可在 Windows 10/11 独立运行的 PyQt6 桌面工具，安全批量替换 DOCX 中的文本，避免重复与级联替换，并保持匹配位置的格式。

**Architecture:** 界面层只收集输入和展示状态，文件发现、纯文本匹配、OOXML/Run 写回、DOCX 遍历与安全保存各自独立。单个 `QThread` 使用冻结的文件和规则快照顺序处理文档，通过信号回传进度，并在文件边界响应取消请求。

**Tech Stack:** CPython 3.11 x64、PyQt6、python-docx、pytest、pytest-qt、PyInstaller、Windows 10/11。

**Spec:** `docs/开发全流程.md`

## Global Constraints

- 首版只处理 `.docx`；`.doc` 在添加阶段提示并计入“已跳过”，不得进入处理队列。
- 首版不依赖 Microsoft Word COM、LibreOffice、Excel、WPS 或其他外部办公软件。
- 文本区域包括正文、普通及嵌套表格、页眉、页脚和超链接显示文字。
- 不处理跨段落匹配、文本框、艺术字、批注、脚注、尾注、自动域或域代码。
- 匹配使用不可变原文快照，从左到右扫描，同位置最长规则优先；替换结果不得再次参与匹配。
- 新文本继承匹配首字符所在 Run 的格式，未匹配内容保持原有 Run 与格式。
- “替换前”和“替换后”各自最多 10,000 个字符；纯空白“替换前”无效；空“替换后”表示删除。
- 单文件最大允许大小为 `50 × 1024 × 1024` 字节；文件夹扫描不递归；规范化绝对路径去重。
- 源文件永不修改；无匹配不生成副本；现有输出永不覆盖。
- 输出先写入目标目录临时 DOCX，重新打开验证成功后再移动为最终名称。
- 主窗口初始 `1000×900`、可缩放、不使用绝对坐标；按钮不得遮挡内容。
- 当前目录尚未初始化 Git；执行 Task 1 时建立本地仓库，后续每个任务通过测试后提交一次。

    results: tuple[FileResult, ...] = ()
    cancelled: bool = False

        return sum(result.status is status for result in self.results)

# src/word_text_tool/replacement_engine.py
from dataclasses import dataclass
from typing import Sequence

from word_text_tool.models import ReplacementRule

MAX_RULE_TEXT_LENGTH = 10_000


class RuleValidationError(ValueError):
    """用户输入的替换规则无效。"""


@dataclass(frozen=True, slots=True)
class TextMatch:
    start: int
    end: int
    replacement: str


@dataclass(frozen=True, slots=True)
class ReplacementOutcome:
    text: str
    count: int


def validate_rule(before: str, after: str, existing_rules: Sequence[ReplacementRule]) -> ReplacementRule:
    if not before or before.isspace():
        raise RuleValidationError("替换前不能为空或纯空白")
    if len(before) > MAX_RULE_TEXT_LENGTH or len(after) > MAX_RULE_TEXT_LENGTH:
        raise RuleValidationError("替换前和替换后均不能超过 10000 个字符")
    if any(rule.before == before for rule in existing_rules):
        raise RuleValidationError("相同的替换前内容已存在")
    return ReplacementRule(before=before, after=after)


def find_matches(text: str, rules: Sequence[ReplacementRule]) -> tuple[TextMatch, ...]:
    ordered = sorted(enumerate(rules), key=lambda item: (-len(item[1].before), item[0]))
    matches: list[TextMatch] = []
    position = 0
    while position < len(text):
        selected = next((rule for _, rule in ordered if text.startswith(rule.before, position)), None)
        if selected is None:
            position += 1
            continue
        end = position + len(selected.before)
        matches.append(TextMatch(position, end, selected.after))
        position = end
    return tuple(matches)


def replace_text(text: str, rules: Sequence[ReplacementRule]) -> ReplacementOutcome:
    matches = find_matches(text, rules)
    if not matches:
        return ReplacementOutcome(text=text, count=0)
    parts: list[str] = []
    cursor = 0
    for match in matches:
        parts.append(text[cursor:match.start])
        parts.append(match.replacement)
        cursor = match.end
    parts.append(text[cursor:])
    return ReplacementOutcome(text="".join(parts), count=len(matches))
```

# 追加到 src/word_text_tool/replacement_engine.py
from docx.text.run import Run


def _locate_start(lengths: Sequence[int], offset: int) -> tuple[int, int]:
    cursor = 0
    for index, length in enumerate(lengths):
        if offset < cursor + length:
            return index, offset - cursor
        cursor += length
    return len(lengths) - 1, lengths[-1]


def _locate_end(lengths: Sequence[int], offset: int) -> tuple[int, int]:
    cursor = 0
    for index, length in enumerate(lengths):
        cursor += length
        if offset <= cursor:
            return index, offset - (cursor - length)
    return len(lengths) - 1, lengths[-1]


def replace_runs(runs: Sequence[Run], rules: Sequence[ReplacementRule]) -> int:
    if not runs:
        return 0
    original_texts = [run.text for run in runs]
    lengths = [len(text) for text in original_texts]
    matches = find_matches("".join(original_texts), rules)
    if not matches:
        return 0
    rewritten = original_texts.copy()
    for match in reversed(matches):
        start_index, start_offset = _locate_start(lengths, match.start)
        end_index, end_offset = _locate_end(lengths, match.end)
        if start_index == end_index:
            current = rewritten[start_index]
            rewritten[start_index] = current[:start_offset] + match.replacement + current[end_offset:]
            continue
        rewritten[start_index] = rewritten[start_index][:start_offset] + match.replacement
        for index in range(start_index + 1, end_index):
            rewritten[index] = ""
        rewritten[end_index] = rewritten[end_index][end_offset:]
    for run, old_text, new_text in zip(runs, original_texts, rewritten, strict=True):
        if old_text != new_text:
            run.text = new_text
    return len(matches)
```

# src/word_text_tool/document_processor.py
from collections.abc import Iterator, Sequence

from docx.document import Document as DocumentObject
from docx.oxml.ns import qn
from docx.text.paragraph import Paragraph
from docx.text.run import Run

from word_text_tool.models import ReplacementRule
from word_text_tool.replacement_engine import replace_runs

_RUN = qn("w:r")
_HYPERLINK = qn("w:hyperlink")
_FIELD_SIMPLE = qn("w:fldSimple")
_FIELD_CHAR = qn("w:fldChar")
_FIELD_CHAR_TYPE = qn("w:fldCharType")
_SAFE_RUN_CHILDREN = {qn("w:rPr"), qn("w:t"), qn("w:tab"), qn("w:br"), qn("w:cr")}


def _iter_container_paragraphs(container) -> Iterator[Paragraph]:
    yield from container.paragraphs
    seen_cells: set[int] = set()
    for table in container.tables:
        for row in table.rows:
            for cell in row.cells:
                identity = id(cell._tc)
                if identity in seen_cells:
                    continue
                seen_cells.add(identity)
                yield from _iter_container_paragraphs(cell)


def _is_safe_text_run(run_element) -> bool:
    children = {child.tag for child in run_element}
    return children.issubset(_SAFE_RUN_CHILDREN) and any(
        child.tag in {qn("w:t"), qn("w:tab"), qn("w:br"), qn("w:cr")}
        for child in run_element
    )


def _field_markers(run_element) -> tuple[str, ...]:
    return tuple(marker.get(_FIELD_CHAR_TYPE, "") for marker in run_element.iter(_FIELD_CHAR))


def _iter_run_groups(paragraph: Paragraph) -> Iterator[tuple[Run, ...]]:
    normal_group: list[Run] = []
    field_depth = 0

    def flush() -> tuple[Run, ...]:
        group = tuple(normal_group)
        normal_group.clear()
        return group

    for child in paragraph._p:
        if child.tag == _FIELD_SIMPLE:
            group = flush()
            if group:
                yield group
            continue
        if child.tag == _RUN:
            markers = _field_markers(child)
            if markers:
                group = flush()
                if group:
                    yield group
                for marker in markers:
                    if marker == "begin":
                        field_depth += 1
                    elif marker == "end":
                        field_depth = max(0, field_depth - 1)
                continue
            if field_depth == 0 and _is_safe_text_run(child):
                normal_group.append(Run(child, paragraph))
            else:
                group = flush()
                if group:
                    yield group
            continue
        group = flush()
        if group:
            yield group
        if child.tag == _HYPERLINK and field_depth == 0:
            hyperlink_runs = tuple(
                Run(run_element, paragraph)
                for run_element in child.iterchildren(tag=_RUN)
                if _is_safe_text_run(run_element)
            )
            if hyperlink_runs:
                yield hyperlink_runs
    group = flush()
    if group:
        yield group


def _replace_paragraph(paragraph: Paragraph, rules: Sequence[ReplacementRule]) -> int:
    return sum(replace_runs(group, rules) for group in _iter_run_groups(paragraph))


def replace_document(document: DocumentObject, rules: Sequence[ReplacementRule]) -> int:
    total = sum(_replace_paragraph(p, rules) for p in _iter_container_paragraphs(document))
    seen_parts: set[str] = set()
    for section in document.sections:
        for story in (section.header, section.footer):
            part_name = str(story.part.partname)
            if part_name in seen_parts:
                continue
            seen_parts.add(part_name)
            total += sum(_replace_paragraph(p, rules) for p in _iter_container_paragraphs(story))
    return total
```

# src/word_text_tool/file_service.py
import os
import tempfile
from collections.abc import Iterable
from pathlib import Path

from word_text_tool.models import DiscoveryResult, FileResult, FileStatus

MAX_FILE_SIZE_BYTES = 50 * 1024 * 1024


class OutputDirectoryError(OSError):
    """输出目录不存在或不可写。"""


def _key(path: Path) -> str:
    return os.path.normcase(str(path.resolve(strict=False)))


def _skip(path: Path, message: str) -> FileResult:
    return FileResult(path.resolve(strict=False), FileStatus.SKIPPED, message)


def discover_files(candidates: Iterable[Path], existing: Iterable[Path] = ()) -> DiscoveryResult:
    accepted: list[Path] = []
    skipped: list[FileResult] = []
    seen = {_key(path) for path in existing}
    expanded: list[Path] = []
    for candidate in candidates:
        path = Path(candidate)
        if path.is_dir():
            expanded.extend(sorted(
                (child for child in path.iterdir() if child.is_file() and child.suffix.lower() in {".doc", ".docx"}),
                key=lambda child: child.name.casefold(),
            ))
        else:
            expanded.append(path)
    for path in expanded:
        resolved = path.resolve(strict=False)
        suffix = resolved.suffix.lower()
        if not resolved.exists() or not resolved.is_file():
            skipped.append(_skip(resolved, "文件不存在或不是普通文件"))
            continue
        if suffix == ".doc":
            skipped.append(_skip(resolved, "旧版 .doc 暂不支持，将在后续版本处理"))
            continue
        if suffix != ".docx":
            skipped.append(_skip(resolved, "不支持的文件格式"))
            continue
        key = _key(resolved)
        if key in seen:
            skipped.append(_skip(resolved, "重复文件已跳过"))
            continue
        if resolved.stat().st_size > MAX_FILE_SIZE_BYTES:
            skipped.append(_skip(resolved, "文件超过 50MB，已跳过"))
            continue
        try:
            with resolved.open("rb") as stream:
                stream.read(1)
        except OSError:
            skipped.append(_skip(resolved, "文件不可读取"))
            continue
        seen.add(key)
        accepted.append(resolved)
    return DiscoveryResult(tuple(accepted), tuple(skipped))


def _assert_writable(directory: Path) -> None:
    try:
        with tempfile.NamedTemporaryFile(dir=directory, prefix=".write-test-", delete=True):
            return
    except OSError as exc:
        raise OutputDirectoryError("保存目录不可写") from exc


def prepare_output_directory(source: Path, custom_output_dir: Path | None) -> Path:
    if custom_output_dir is None:
        directory = source.resolve().parent / "修改版"
        directory.mkdir(parents=False, exist_ok=True)
    else:
        directory = custom_output_dir.resolve(strict=False)
        if not directory.exists() or not directory.is_dir():
            raise OutputDirectoryError("自定义保存目录不存在")
    _assert_writable(directory)
    return directory


def next_output_path(source: Path, output_dir: Path) -> Path:
    base = f"{source.stem} - 修改版"
    candidate = output_dir / f"{base}.docx"
    index = 1
    while candidate.exists():
        candidate = output_dir / f"{base} ({index}).docx"
        index += 1
    return candidate
```

import os
import tempfile
from pathlib import Path

from docx import Document
from docx.opc.exceptions import PackageNotFoundError

from word_text_tool.file_service import next_output_path, prepare_output_directory
from word_text_tool.models import FileResult, FileStatus


class DocumentProcessor:
    def process(self, source: Path, rules: Sequence[ReplacementRule], custom_output_dir: Path | None) -> FileResult:
        source = source.resolve(strict=False)
        temporary_path: Path | None = None
        try:
            document = Document(source)
            replacement_count = replace_document(document, rules)
            if replacement_count == 0:
                return FileResult(source, FileStatus.UNMATCHED, "未找到匹配内容")
            output_dir = prepare_output_directory(source, custom_output_dir)
            with tempfile.NamedTemporaryFile(dir=output_dir, prefix=".word-text-tool-", suffix=".docx", delete=False) as temporary:
                temporary_path = Path(temporary.name)
            document.save(temporary_path)
            verified = Document(temporary_path)
            verified._element.getroottree()
            while True:
                output_path = next_output_path(source, output_dir)
                try:
                    os.rename(temporary_path, output_path)
                    break
                except FileExistsError:
                    continue
            temporary_path = None
            return FileResult(source, FileStatus.SUCCESS, "修改完成，已保存", output_path, replacement_count)
        except PackageNotFoundError:
            return FileResult(source, FileStatus.FAILED, "文档损坏或不是有效的 DOCX 文件")
        except PermissionError:
            return FileResult(source, FileStatus.FAILED, "文件被占用或没有访问权限")
        except OSError as exc:
            return FileResult(source, FileStatus.FAILED, f"文件处理失败：{exc}")
        except Exception as exc:
            return FileResult(source, FileStatus.FAILED, f"文档处理失败：{exc}")
        finally:
            if temporary_path is not None:
                temporary_path.unlink(missing_ok=True)

# src/word_text_tool/batch_worker.py
from pathlib import Path
from threading import Event

from PyQt6.QtCore import QThread, pyqtSignal

from word_text_tool.document_processor import DocumentProcessor
from word_text_tool.models import BatchSummary, ReplacementRule


class BatchWorker(QThread):
    progress = pyqtSignal(int, int, str, int)
    batch_finished = pyqtSignal(object)

    def __init__(self, processor: DocumentProcessor, files: tuple[Path, ...], rules: tuple[ReplacementRule, ...], custom_output_dir: Path | None, parent=None):
        super().__init__(parent)
        self._processor = processor
        self._files = tuple(files)
        self._rules = tuple(rules)
        self._custom_output_dir = custom_output_dir
        self._cancel_requested = Event()

    def request_cancel(self) -> None:
        self._cancel_requested.set()

    def run(self) -> None:
        results = []
        total = len(self._files)
        for index, source in enumerate(self._files, start=1):
            if self._cancel_requested.is_set():
                break
            results.append(self._processor.process(source, self._rules, self._custom_output_dir))
            self.progress.emit(index, total, source.name, round(index * 100 / total))
        self.batch_finished.emit(BatchSummary(tuple(results), self._cancel_requested.is_set()))

def _start_processing(self):
    files = self.file_panel.files()
    rules = self.rule_panel.rules()
    if not files:
        QMessageBox.warning(self, "无法开始", "请至少选择一个有效的 DOCX 文件")
        return
    if not rules:
        QMessageBox.warning(self, "无法开始", "请至少添加一条替换规则")
        return
    if self.custom_output_dir is not None:
        try:
            prepare_output_directory(files[0], self.custom_output_dir)
        except OutputDirectoryError as exc:
            QMessageBox.warning(self, "保存路径不可用", str(exc))
            return
    self._set_processing_state(True)
    self.progress_bar.setValue(0)
    self.worker = self.worker_factory(
        DocumentProcessor(), tuple(files), tuple(rules), self.custom_output_dir, self
    )
    self.worker.progress.connect(self._on_progress)
    self.worker.batch_finished.connect(self._on_batch_finished)
    self.worker.start()

def _cancel_processing(self):
    if self.worker is not None and self.worker.isRunning():
        self.status_label.setText("正在安全结束当前文件…")
        self.worker.request_cancel()
        self.cancel_button.setEnabled(False)

def _on_batch_finished(self, worker_summary):
    results = tuple(self.file_panel.skipped_results()) + worker_summary.results
    summary = BatchSummary(results, worker_summary.cancelled)
    self._set_processing_state(False)
    self.status_label.setText("处理已取消" if summary.cancelled else "处理完成")
    ResultDialog(summary, self).exec()
    self.worker.deleteLater()
    self.worker = None
    if self._close_after_worker:
        self.close()

# src/word_text_tool/app.py
import sys

from PyQt6.QtGui import QFont
from PyQt6.QtWidgets import QApplication

from word_text_tool.ui.main_window import MainWindow


def create_application(argv=None):
    application = QApplication.instance() or QApplication(argv if argv is not None else sys.argv)
    application.setApplicationName("Word 文本处理工具")
    application.setOrganizationName("WordTextTool")
    application.setFont(QFont("Microsoft YaHei UI", 10))
    return application


def run() -> int:
    application = create_application()
    window = MainWindow()
    window.show()
    return application.exec()
